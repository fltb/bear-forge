import test from 'node:test';
import assert from 'node:assert/strict';
import { z } from 'zod';
import { ExecutionRefSchema, BoundaryIdSchema, CompiledGameDataSchema, GameRunOptionsSchema } from '@bear-forge/contracts';
import type { GameUpdate, GameQueryView, ExecutionView, GameRequestPacket, GameResultPacket, ManagedStateCapability } from '@bear-forge/contracts';
import { module, schemas, mappedProgram } from './authoring.ts';
import type { Example, Request, Owned } from './authoring.ts';
import { executionPeer } from '../support/execution-peer.ts';

const packetSchemas={
  start:schemas.setup,
  request:z.strictObject({state:schemas.state,request:z.strictObject({type:z.literal('add'),data:z.null()}),events:z.array(schemas.event)}),
  reply:z.strictObject({type:z.literal('add'),data:schemas.interactions.add.input}),
  result:z.strictObject({state:schemas.state,result:schemas.result,events:z.array(schemas.event)}),
};
// This native fixture checks the protocol mapping, not managed continuation persistence.
const mappedPeer=()=>executionPeer(async (setup:Example['setup'],core)=>{
  let owned:Owned|undefined;
  const state:ManagedStateCapability<Owned>={
    initialize:initial=>{assert.equal(owned,undefined);owned=structuredClone(initial);},
    read:()=>{assert.ok(owned);return structuredClone(owned);},
    write:next=>{assert.ok(owned);owned=structuredClone(next);},
  };
  return mappedProgram(setup,{...core,state});
},packetSchemas);
const asGame=(view:ExecutionView<GameRequestPacket<Example>,GameResultPacket<Example>>):GameUpdate<Request,Example['event'],Example['result']>=>{
  const b=view.boundary;
  if(b.kind==='fault')throw new Error(`fault:${b.error.code}`);
  if(b.kind==='done')return {boundary:{kind:'ended',result:b.result.result},events:b.result.events};
  return {boundary:{kind:'waiting',boundaryId:BoundaryIdSchema.parse(b.requestId),pending:b.request.request},events:b.request.events};
};
const queryView=(view:ExecutionView<GameRequestPacket<Example>,GameResultPacket<Example>>):GameQueryView<Example>=>{
  const b=view.boundary;
  if(b.kind==='fault')throw new Error('fault is not an author query boundary');
  return b.kind==='waiting'?{state:b.request.state,boundary:{kind:'waiting',request:b.request.request}}:{state:b.result.state,boundary:{kind:'ended',result:b.result.result}};
};

test('author module maps to existing Core/Game: events drain once and rejected input preserves boundary',async()=>{
  const core=mappedPeer(),started=await core.start({initial:10});
  assert.ok(started.ok);if(!started.ok)throw new Error();
  let view=started.value;
  assert.deepEqual(asGame(view).events,[]);
  assert.equal(module.queries.observe.run(queryView(view),null),10);
  const advance=async (input:number)=>{
    const b=view.boundary;if(b.kind!=='waiting')throw new Error('not waiting');
    const check=module.inputs.add.validate({state:b.request.state,request:b.request.request.data,input});
    if(!check.valid)return check;
    const next=await core.resume({execution:view.execution,requestId:b.requestId,reply:{type:'add',data:input}});
    assert.ok(next.ok);if(!next.ok)throw new Error();view=next.value;return check;
  };
  const before=await core.read({execution:view.execution,afterSequence:null,limit:20});
  assert.deepEqual(await advance(-1),{valid:false,reason:'outside_range'});
  assert.deepEqual(await core.inspect(view.execution),started);
  assert.deepEqual(await core.read({execution:view.execution,afterSequence:null,limit:20}),before);
  await advance(2);
  assert.deepEqual(asGame(view).events,[{kind:'added',value:2}]);
  assert.equal(module.queries.observe.run(queryView(view),null),12);
  await advance(1);
  assert.deepEqual(asGame(view),{boundary:{kind:'ended',result:13},events:[{kind:'added',value:1}]});
  assert.equal(module.queries.observe.run(queryView(view),null),13);
  // Inspection repeats a boundary, not a new emission/transition.
  assert.deepEqual(await core.inspect(view.execution),{ok:true,value:view});
});
test('one author function also runs natively with matching state, requests, events and result',async()=>{
  let state=module.initialize({initial:10});const inputs=[2,1],requests:Request[]=[],events:Example['event'][]=[];
  const result=await module.run({
    state:{read:()=>structuredClone(state),write:next=>{state=structuredClone(next);}},
    input:{request:async (type,data)=>{requests.push(structuredClone({type,data}));return inputs.shift()!;}},
    output:{emit:event=>events.push(structuredClone(event))},
    random:{integer:async()=>{throw new Error('this game does not request randomness');}},
  });
  const core=mappedPeer();let current=await core.start({initial:10});
  const outputs:ReturnType<typeof asGame>[]=[],pending:Request[]=[];
  for(const input of [2,1]){
    assert.ok(current.ok);if(!current.ok||current.value.boundary.kind!=='waiting')throw new Error();
    outputs.push(asGame(current.value));pending.push(current.value.boundary.request.request);
    current=await core.resume({execution:current.value.execution,requestId:current.value.boundary.requestId,reply:{type:'add',data:input}});
  }
  assert.ok(current.ok);if(!current.ok||current.value.boundary.kind!=='done')throw new Error();
  outputs.push(asGame(current.value));
  assert.deepEqual(current.value.boundary.result.state,state);
  assert.equal(current.value.boundary.result.result,result);
  assert.deepEqual(pending,requests);assert.deepEqual(outputs.flatMap(x=>x.events),events);
});
test('artifact shape is separate from source modules and is not an admission proof',()=>{
  assert.equal(CompiledGameDataSchema.safeParse(module).success,false);
  assert.equal(CompiledGameDataSchema.safeParse({kind:'controlled-game',content:'javascript'}).success,false);
  assert.equal(CompiledGameDataSchema.safeParse({kind:'controlled-game',content:new Uint8Array([0])}).success,true);
  assert.equal(GameRunOptionsSchema.safeParse({randomSeed:new Uint8Array([1])}).success,true);
  // Byte shape acceptance intentionally makes no claim about bytecode validity or authorization.
});
test('Core faults cannot become a Game normal ending',()=>{
  const view:ExecutionView<GameRequestPacket<Example>,GameResultPacket<Example>>={execution:ExecutionRefSchema.parse({executionId:'00000000-0000-4000-8000-000000000000'}),boundary:{kind:'fault',error:{kind:'fault',code:'program_failed',message:'failed'}}};
  assert.throws(()=>asGame(view),/fault:program_failed/);
  assert.throws(()=>queryView(view),/fault/);
});
