import test from 'node:test';
import assert from 'node:assert/strict';
import { z } from 'zod';
import { CoreErrorSchema, ExecutionStatusSchema, GameCapabilitySchema, ExecutionRefSchema, RequestIdSchema, RandomIntegerInputSchema } from '@bear-forge/contracts';
import type { Program, RandomCapability, ProgramSchemas, StateTransitionCapability } from '@bear-forge/contracts';
import { executionPeer } from '../support/execution-peer.ts';

test('fixed categories are enums; opaque identifiers have concrete validation',()=>{
  assert.equal(ExecutionStatusSchema.safeParse('landlord').success,false);
  assert.equal(CoreErrorSchema.safeParse({kind:'fault',code:'invented',message:'x'}).success,false);
  assert.equal(GameCapabilitySchema.safeParse('opponents').success,false);
  assert.equal(ExecutionRefSchema.safeParse({executionId:'anything'}).success,false);
  assert.equal(RandomIntegerInputSchema.safeParse({minInclusive:1,maxExclusive:1}).success,false);
  assert.equal(RandomIntegerInputSchema.safeParse({minInclusive:0.2,maxExclusive:1}).success,false);
});
test('generic Core requests/capture work without game concepts; invalid replies do not advance',async()=>{
  const schemas:ProgramSchemas<number,string,number,number>={start:z.number(),request:z.string(),reply:z.number(),result:z.number()};
  const program:Program<number,string,number,number,RandomCapability>=async (n,c)=>{
    let sum=n;for(let i=0;i<2;i++)sum+=await c.request(`number-${i}`);return sum;
  };
  const peer=executionPeer(program,schemas),started=await peer.start(1);assert.ok(started.ok);if(!started.ok||started.value.boundary.kind!=='waiting')throw new Error();
  const a=started.value,request=a.boundary;if(request.kind!=='waiting')throw new Error();
  assert.deepEqual(await peer.inspect(a.execution),started);
  const before=await peer.read({execution:a.execution,afterSequence:null,limit:10});
  assert.equal((await peer.resume({execution:a.execution,requestId:RequestIdSchema.parse('00000000-0000-4000-8000-000000000001'),reply:2})).ok,false);
  assert.equal((await peer.resume({execution:a.execution,requestId:request.requestId,reply:'bad' as unknown as number})).ok,false);
  assert.deepEqual(await peer.read({execution:a.execution,afterSequence:null,limit:10}),before);
  const second=await peer.resume({execution:a.execution,requestId:request.requestId,reply:2});if(!second.ok||second.value.boundary.kind!=='waiting')throw new Error();
  const last=await peer.resume({execution:a.execution,requestId:second.value.boundary.requestId,reply:3});assert.ok(last.ok);if(!last.ok)throw new Error();assert.deepEqual(last.value.boundary,{kind:'done',result:6});
  const captured=await peer.read({execution:a.execution,afterSequence:null,limit:10});if(!captured.ok)throw new Error();
  assert.deepEqual(captured.value.records.map(r=>r.kind),['started','requested','resumed','requested','resumed','completed']);
  assert.deepEqual(captured.value.records.map(r=>r.sequence),[0,1,2,3,4,5]);
});
test('same transition capability supports stack exploration and repeated rollouts',async()=>{
  const model:StateTransitionCapability<readonly number[],number,number>={
    inspect:async state=>state.reduce((a,b)=>a+b,0),
    transition:async ({state,input})=>({state:[...state,input],view:state.reduce((a,b)=>a+b,0)+input}),
  };
  const root:readonly number[]=[];
  const a=await model.transition({state:root,input:2}),b=await model.transition({state:root,input:3});
  assert.deepEqual(root,[]);assert.deepEqual(a.state,[2]);assert.deepEqual(b.state,[3]);
  const deeper=await model.transition({state:a.state,input:4});assert.equal(deeper.view,6);assert.deepEqual(a.state,[2]);
  assert.deepEqual(await model.transition({state:root,input:2}),a);
});
