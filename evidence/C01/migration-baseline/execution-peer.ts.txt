/** Test peer for the public Execution/Capture contracts. Runs the real program on V8.
 * No persistence, compiler, wall-time scheduler, or production Core is implemented here. */
import { randomUUID } from 'node:crypto';
import { ExecutionRefSchema, RequestIdSchema, CaptureReadInputSchema, RandomIntegerInputSchema } from '@bear-forge/contracts';
import type { Execution, ExecutionCapture, ExecutionView, ExecutionRecord, ExecutionRef, CoreCall, CoreError, CoreFault, Program, ProgramSchemas, RandomCapability } from '@bear-forge/contracts';

export const executionPeer = <S,Q,A,R>(program:Program<S,Q,A,R,RandomCapability>,schemas:ProgramSchemas<S,Q,A,R>,seed=11):Execution<S,Q,A,R>&ExecutionCapture<S,Q,A,R> => {
  type Record = ExecutionRecord<S,Q,A,R>;
  type Entry = {ref:ExecutionRef;view:ExecutionView<Q,R>|null;records:Record[];rng:number;closed:boolean;reply:((value:A)=>void)|null;cancel:((error:Error)=>void)|null;notify:((view:ExecutionView<Q,R>)=>void)|null};
  const entries=new Map<string,Entry>();
  const success=<T>(value:T):CoreCall<T>=>({ok:true,value});
  const failure=<T>(code:Extract<CoreError,{kind:'rejected'|'conflict'}>['code']):CoreCall<T>=>{
    if(code==='execution_closed'||code==='execution_busy'||code==='request_mismatch'||code==='snapshot_incompatible')return {ok:false,error:{kind:'conflict',code,message:code}};
    return {ok:false,error:{kind:'rejected',code,message:code}};
  };
  const publish=(e:Entry,view:ExecutionView<Q,R>)=>{if(e.closed)return;e.view=structuredClone(view);const notify=e.notify;e.notify=null;notify?.(structuredClone(view));};
  const next=(e:Entry)=>new Promise<ExecutionView<Q,R>>(resolve=>{e.notify=resolve;});
  const emit=(e:Entry,r:Record)=>e.records.push(structuredClone(r));
  const fault=(e:Entry,error:CoreFault)=>{
    if(e.closed||e.view?.boundary.kind==='fault'||e.view?.boundary.kind==='done')return;
    const cancel=e.cancel;e.cancel=null;e.reply=null;
    emit(e,{sequence:e.records.length,kind:'faulted',error});
    publish(e,{execution:e.ref,boundary:{kind:'fault',error}});
    cancel?.(new Error(error.message));
  };
  const running=(e:Entry)=>{
    if(e.closed)throw new Error('execution closed');
    if(e.view!==null){
      fault(e,{kind:'fault',code:'program_failed',message:'effect outside running phase'});
      throw new Error('effect outside running phase');
    }
  };
  return {
    async start(raw){
      const parsed=schemas.start.safeParse(raw);if(!parsed.success)return failure('invalid_input');
      const ref=ExecutionRefSchema.parse({executionId:randomUUID()});
      const e:Entry={ref,view:null,records:[],rng:seed,closed:false,reply:null,cancel:null,notify:null};entries.set(ref.executionId,e);
      emit(e,{sequence:0,kind:'started',input:parsed.data});
      const waiting=next(e);
      void Promise.resolve().then(()=>program(parsed.data,{
        async integer({minInclusive,maxExclusive}){
          running(e);
          if(!RandomIntegerInputSchema.safeParse({minInclusive,maxExclusive}).success){fault(e,{kind:'fault',code:'program_failed',message:'invalid random bounds'});throw new Error('invalid random bounds');}
          e.rng=(Math.imul(e.rng,1664525)+1013904223)>>>0;
          return minInclusive+e.rng%(maxExclusive-minInclusive);
        },
        async request(rawRequest){
          running(e);
          const checked=schemas.request.safeParse(rawRequest);
          if(!checked.success){fault(e,{kind:'fault',code:'invalid_output',message:'invalid request'});throw new Error('invalid request');}
          const request=checked.data,requestId=RequestIdSchema.parse(randomUUID());
          const reply=new Promise<A>((resolve,reject)=>{e.reply=resolve;e.cancel=reject;});
          emit(e,{sequence:e.records.length,kind:'requested',requestId,request});
          publish(e,{execution:ref,boundary:{kind:'waiting',requestId,request}});
          return reply;
        },
      })).then(rawResult=>{
        if(e.closed||e.view?.boundary.kind==='fault')return;
        if(e.reply){fault(e,{kind:'fault',code:'program_failed',message:'program returned with pending request'});return;}
        const checked=schemas.result.safeParse(rawResult);
        if(!checked.success){fault(e,{kind:'fault',code:'invalid_output',message:'invalid result'});return;}
        const result=checked.data;
        emit(e,{sequence:e.records.length,kind:'completed',result});publish(e,{execution:ref,boundary:{kind:'done',result}});
      }).catch(()=>{
        fault(e,{kind:'fault',code:'program_failed',message:'test peer program failed'});
      });
      return success(await waiting);
    },
    async resume({execution,requestId,reply:raw}){
      const e=entries.get(execution.executionId);if(!e)return failure('execution_not_found');
      if(e.closed)return failure('execution_closed');
      if(e.view===null)return failure('execution_busy');
      if(e.view?.boundary.kind!=='waiting'||e.view.boundary.requestId!==requestId||!e.reply)return failure('request_mismatch');
      const parsed=schemas.reply.safeParse(raw);if(!parsed.success)return failure('invalid_input');
      const waiting=next(e),reply=e.reply;e.reply=null;e.cancel=null;e.view=null;
      emit(e,{sequence:e.records.length,kind:'resumed',requestId,reply:parsed.data});reply(parsed.data);
      return success(await waiting);
    },
    async inspect(ref){const e=entries.get(ref.executionId);return !e?failure('execution_not_found'):e.closed?failure('execution_closed'):e.view===null?failure('execution_busy'):success(structuredClone(e.view));},
    async close(ref){const e=entries.get(ref.executionId);if(!e)return failure('execution_not_found');if(e.closed)return success(undefined);if(e.view===null)return failure('execution_busy');e.closed=true;e.cancel?.(new Error('closed'));e.cancel=null;e.reply=null;e.records=[];return success(undefined);},
    async read(raw){
      const parsed=CaptureReadInputSchema.safeParse(raw);if(!parsed.success)return failure('invalid_input');
      const {execution,afterSequence,limit}=parsed.data;
      const e=entries.get(execution.executionId);if(!e)return failure('execution_not_found');
      if(e.closed)return failure('execution_closed');
      if(afterSequence!==null&&afterSequence>=e.records.length)return failure('invalid_input');
      const start=(afterSequence??-1)+1,records=e.records.slice(start,start+limit);
      return success({records:structuredClone(records),nextSequence:start+records.length<e.records.length?records.at(-1)!.sequence:null});
    },
  };
};
