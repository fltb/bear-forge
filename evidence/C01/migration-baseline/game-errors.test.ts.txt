import test from 'node:test';
import assert from 'node:assert/strict';
import { GameErrorSchema, CoreErrorSchema } from '@bear-forge/contracts';
import { fromCore } from '../../games/doudizhu/src/call-errors.ts';
import { assertBoundarySchema, parseBoundary } from '../../tools/schema_admission.ts';

test('Game failures have closed kind/code/state combinations and value-preserving schemas',()=>{
  assertBoundarySchema(GameErrorSchema);
  for(const error of [
    {kind:'rejected',code:'invalid_input',message:''},
    {kind:'conflict',code:'game_busy',message:''},
    {kind:'unsupported',code:'capability_unavailable',message:''},
    {kind:'fault',code:'program_failed',state:'unchanged',message:''},
    {kind:'fault',code:'execution_uncertain',state:'unavailable',message:''},
  ])assert.deepEqual(parseBoundary(GameErrorSchema,error),error);
  for(const error of [
    {kind:'rejected',code:'program_failed',message:''},
    {kind:'fault',code:'execution_uncertain',state:'unchanged',message:''},
    {kind:'fault',code:'game_faulted',state:'unchanged',message:''},
    {kind:'fault',code:'program_failed',message:''},
    {kind:'conflict',code:'game_busy',state:'unavailable',message:''},
    {kind:'rejected',code:'custom_game_rule',message:''},
  ])assert.equal(GameErrorSchema.safeParse(error).success,false);
});

test('every Core failure maps into the public Game carrier without losing its class',()=>{
  const codes={rejected:['invalid_input','execution_not_found','snapshot_not_found'],conflict:['execution_busy','execution_closed','request_mismatch','snapshot_incompatible'],unsupported:['capability_unavailable'],fault:['invalid_output','budget_exceeded','program_failed']};
  for(const [kind,items] of Object.entries(codes))for(const code of items){
    const core=CoreErrorSchema.parse({kind,code,message:'evidence'});
    const game=GameErrorSchema.parse(fromCore(core));
    assert.equal(game.kind,core.kind);assert.equal(game.message,core.message);
    if(game.kind==='fault')assert.equal(game.state,'unavailable');
  }
});
