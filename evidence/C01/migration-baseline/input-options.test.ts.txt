import test from 'node:test';
import assert from 'node:assert/strict';
import { z } from 'zod';
import { InputOptionsSchema, InputOptionsKindSchema, GameCapabilitySchema } from '@bear-forge/contracts';
import type { InputOptions, InputOptionsCapability, GameRef } from '@bear-forge/contracts';
import { LegalActionsSchema } from '../../games/doudizhu/src/schemas.ts';

// The game supplies this schema; the generic protocol does not interpret the convention.
const DescriptionSchema=z.strictObject({convention:z.literal('bounded-total'),total:z.number().int().nonnegative()});
const ActionSchema=z.strictObject({left:z.number().int(),right:z.number().int()});
const OptionsSchema=z.discriminatedUnion('kind',[
  z.strictObject({kind:z.literal('exact'),values:z.array(ActionSchema)}),
  z.strictObject({kind:z.literal('construct'),description:DescriptionSchema}),
]);
type Action=z.infer<typeof ActionSchema>;
type Description=z.infer<typeof DescriptionSchema>;
const construct:InputOptions<Action,Description>={kind:'construct',description:{convention:'bounded-total',total:7}};
const exact:InputOptions<Action,Description>={kind:'exact',values:[{left:3,right:4}]};

test('exact and construct share a JSON envelope while games own concrete schemas',()=>{
  for(const x of [construct,exact]){
    assert.deepEqual(InputOptionsSchema.parse(JSON.parse(JSON.stringify(x))),x);
    assert.deepEqual(OptionsSchema.parse(x),x);
  }
  assert.equal(InputOptionsKindSchema.safeParse('invented').success,false);
  assert.equal(OptionsSchema.safeParse({kind:'construct',description:{convention:'wrong',total:7}}).success,false);
  assert.equal(InputOptionsSchema.safeParse({kind:'construct',description:()=>1}).success,false);
  assert.equal(InputOptionsSchema.safeParse({kind:'construct',description:undefined}).success,false);
  assert.equal(InputOptionsSchema.safeParse({kind:'exact',values:[NaN]}).success,false);
  assert.deepEqual(InputOptionsSchema.parse({kind:'exact',values:[]}),{kind:'exact',values:[]});
  assert.equal(GameCapabilitySchema.safeParse('input_options').success,true);
});
test('Dou Dizhu only accepts the concrete exact action reply',()=>{
  const reply={kind:'exact',values:[{kind:'bid',value:0},{kind:'bid',value:3}]};
  assert.deepEqual(LegalActionsSchema.parse(reply),reply);
  assert.equal(LegalActionsSchema.safeParse(construct).success,false);
  assert.equal(LegalActionsSchema.safeParse({kind:'exact',values:[{kind:'bid',value:4}]}).success,false);
});
test('construct convention is interpreted by a caller, with no Core or game-state access',()=>{
  // Application-side convention adapter; not a generic construction language.
  const caller=(description:Description):Action=>({left:0,right:description.total});
  assert.deepEqual(caller(DescriptionSchema.parse(construct.description)),{left:0,right:7});
});

export const typeChecks=(ref:GameRef,options:InputOptionsCapability<null,Action,Description>)=>{
  options.describe({target:{kind:'instance',game:ref},context:null});
  // @ts-expect-error construction descriptors are JSON, not executable functions
  const invalid:InputOptions<Action,()=>number>={kind:'construct',description:()=>1};
  // @ts-expect-error an exact-only provider cannot return a construction description
  const invalidExact:InputOptions<Action>={kind:'construct',description:{convention:'bounded-total',total:7}};
  // @ts-expect-error exact values must match the concrete action type
  const invalidValue:InputOptions<Action>={kind:'exact',values:[{left:'wrong',right:2}]};
  return [invalid,invalidExact,invalidValue];
};
