/** A game-owned finite convention; no generic construction interpreter. */
import test from 'node:test';
import assert from 'node:assert/strict';
import { z } from 'zod';
import type { GameModule } from '@bear-forge/contracts';
const Actor=z.enum(['a','b']);
const Reply=z.strictObject({actor:Actor,at:z.number().int(),amount:z.number().int()});
const Context=z.strictObject({actor:Actor,at:z.number().int()});
const Description=z.strictObject({convention:z.literal('allocation-range'),actor:Actor,at:z.number().int(),maximum:z.number().int().nonnegative()});
const schemas={setup:z.strictObject({a:z.number().int().min(0).max(3),b:z.number().int().min(0).max(3)}),state:z.strictObject({a:z.number().int().min(0).max(3),b:z.number().int().min(0).max(3)}),event:z.never(),result:Reply,interactions:{allocate:{request:z.null(),input:Reply,context:Context,description:Description}}};
type G={setup:z.infer<typeof schemas.setup>;state:z.infer<typeof schemas.state>;event:never;result:z.infer<typeof Reply>;interactions:{allocate:{request:null;input:z.infer<typeof Reply>;context:z.infer<typeof Context>;description:z.infer<typeof Description>}}};
const module:GameModule<G,{}>={
  schemas,initialize:s=>({...s}),run:async ctx=>await ctx.input.request('allocate',null),queries:{},
  inputs:{allocate:{
    options:z.discriminatedUnion('kind',[z.strictObject({kind:z.literal('exact'),values:z.array(Reply)}),z.strictObject({kind:z.literal('construct'),description:Description})]),
    describe:({state,context})=>({kind:'exact',values:context.at>=0&&context.at<=2?Array.from({length:state[context.actor]},(_,i)=>({...context,amount:i+1})):[]}),
    validate:({state,input})=>input.at>=0&&input.at<=2&&input.amount>=1&&input.amount<=state[input.actor]?{valid:true}:{valid:false,reason:'out of range'},
  }},
};
const signature=(a:G['result'])=>`${a.actor}/${a.at}/${a.amount}`;
// Independent finite domain oracle; it does not call describe or validate.
const universe=()=>['a','b'].flatMap(actor=>Array.from({length:5},(_,t)=>Array.from({length:5},(_,n)=>({actor,at:t-1,amount:n}))).flat());

test('context scopes exact lists, while every listed reply carries all submission fields',()=>{
  for(let a=0;a<=3;a++)for(let b=0;b<=3;b++){
    const state=module.initialize({a,b});
    for(const actor of Actor.options)for(let at=-1;at<=3;at++){
      const context={actor,at},result=module.inputs.allocate.describe({state,request:null,context});
      assert.equal(result.kind,'exact');if(result.kind!=='exact')throw new Error();
      const expected=universe().filter(x=>x.actor===actor&&x.at===at&&x.at>=0&&x.at<=2&&x.amount>=1&&x.amount<=state[actor]).map(x=>`${x.actor}/${x.at}/${x.amount}`);
      assert.deepEqual(result.values.map(signature),expected);
      assert.equal(new Set(result.values.map(signature)).size,result.values.length);
      for(const reply of result.values){
        assert.deepEqual(Reply.parse(reply),reply);
        // No context argument is needed to validate a complete reply.
        assert.equal(module.inputs.allocate.validate({state,request:null,input:reply}).valid,true);
      }
      assert.deepEqual(module.inputs.allocate.describe({state,request:null,context}),result);
    }
    const all=Actor.options.flatMap(actor=>[0,1,2].flatMap(at=>{
      const x=module.inputs.allocate.describe({state,request:null,context:{actor,at}});
      if(x.kind!=='exact')throw new Error();return x.values;
    }));
    const legal=universe().filter(x=>x.at>=0&&x.at<=2&&x.amount>=1&&x.amount<=state[x.actor as 'a'|'b']);
    assert.deepEqual(new Set(all.map(signature)),new Set(legal.map(x=>`${x.actor}/${x.at}/${x.amount}`)));
  }
});

test('a declared construct convention can denote exactly the same scoped set without Core knowledge',()=>{
  const interpret=(description:z.infer<typeof Description>)=>Array.from({length:description.maximum},(_,i)=>({actor:description.actor,at:description.at,amount:i+1}));
  for(const actor of Actor.options)for(let amount=0;amount<=3;amount++)for(let at=-1;at<=3;at++){
    const state={a:amount,b:amount};
    const exact=module.inputs.allocate.describe({state,request:null,context:{actor,at}});
    if(exact.kind!=='exact')throw new Error();
    const description=Description.parse({convention:'allocation-range',actor,at,maximum:at>=0&&at<=2?amount:0});
    assert.deepEqual(interpret(description),exact.values);
  }
});
