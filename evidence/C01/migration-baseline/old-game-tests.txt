async function first(seed=11){const host=make(seed);const created=await host.game.runtime.create(setup());return {...host,ref:created.game,update:created.update};}
async function choose(g:ReturnType<typeof bindDouDizhu>,ref:GameRef,p:Pending):Promise<Input>{
  if(p.phase==='bidding')return action(p,{kind:'bid',value:3});
  if(p.phase==='doubling')return action(p,{kind:'double',value:false});
  if(p.phase==='redoubling')return action(p,{kind:'redouble',value:true});
  const options=await g.options.describe({target:{kind:'instance',game:ref},context:context(p)});
  return action(p,options.values.at(-1)!);
}
async function run(seed=11,seat:Seat='0',override?:(p:Pending,i:number)=>Input|null){
  const {peer,game}=make(seed),start=await game.runtime.create(setup(seat));let update=start.update,i=0;
  const updates=[update];
  while(update.boundary.kind==='waiting'){
    assert.ok(++i<600);const p=update.boundary.pending,input=override?.(p,i)??await choose(game,start.game,p);
    const previous=update.boundary;
    update=await game.runtime.advance({game:start.game,boundaryId:previous.boundaryId,input});updates.push(update);
    await assert.rejects(game.runtime.advance({game:start.game,boundaryId:previous.boundaryId,input}));
  }
  const read=await peer.read({execution:ExecutionRefSchema.parse({executionId:start.game.gameId}),afterSequence:null,limit:2000});assert.ok(read.ok);
  if(!read.ok)throw new Error();
  return {game,ref:start.game,updates,result:update.boundary.result,records:read.value.records};
}

for(const mode of ['reject','malformed','throw','outer-fault','boundary-fault'] as const)test(`executor ${mode} preserves the correct adapter state`,async()=>{
  const peer=executionPeer(program,programSchemas,11);let once=true;
  const game=bindDouDizhu({...peer,resume:async input=>{
    if(!once)return peer.resume(input);once=false;
    if(mode==='reject')return {ok:false,error:{kind:'conflict',code:'request_mismatch',message:'not advanced'}};
    if(mode==='throw')throw new Error('transport lost');
    if(mode==='outer-fault')return {ok:false,error:{kind:'fault',code:'budget_exceeded',message:'budget exhausted'}};
    if(mode==='boundary-fault')return {ok:true,value:{execution:input.execution,boundary:{kind:'fault',error:{kind:'fault',code:'program_failed',message:'program failed'}}}};
    const result=await peer.resume(input);
    if(result.ok&&result.value.boundary.kind==='waiting')result.value.boundary.request.state.phase='ended';
    return result;
  }});
  const {game:ref,update}=await game.runtime.create(setup());
  if(update.boundary.kind!=='waiting')throw new Error();
  const input={game:ref,boundaryId:update.boundary.boundaryId,input:action(update.boundary.pending,{kind:'bid',value:3})};
  await assert.rejects(game.runtime.advance(input),error=>{
    const e=GameErrorSchema.parse(error);
    if(mode==='reject')assert.equal(e.kind,'conflict');
    else {assert.equal(e.kind,'fault');if(e.kind==='fault')assert.equal(e.state,'unavailable');}
    assert.equal(e.code,({reject:'boundary_mismatch',malformed:'invalid_output',throw:'execution_uncertain','outer-fault':'budget_exceeded','boundary-fault':'program_failed'} as const)[mode]);
    return true;
  });
  if(mode==='reject'){
    assert.deepEqual(await game.runtime.inspect(ref),update.boundary);
    assert.equal((await game.runtime.advance(input)).boundary.kind,'waiting');
  }else {
    const faulted=(error:unknown)=>{const e=GameErrorSchema.parse(error);assert.equal(e.kind,'fault');assert.equal(e.code,'game_faulted');return true;};
    await assert.rejects(game.runtime.inspect(ref),faulted);
    await assert.rejects(game.runtime.advance(input),faulted);
    await assert.rejects(game.observation.observe({target:{kind:'instance',game:ref},observer:'0'}),faulted);
  }
  await game.runtime.close(ref);
});

test('close and advance cannot race; failed cleanup remains retryable',async()=>{
  const peer=executionPeer(program,programSchemas,11);
  let release:()=>void=()=>{};
  const gate=new Promise<void>(resolve=>{release=resolve;});
  let once=true;
  const game=bindDouDizhu({...peer,close:async ref=>{
    if(once){once=false;await gate;throw new Error('close receipt lost');}
    return peer.close(ref);
  }});
  const {game:ref,update}=await game.runtime.create(setup());
  if(update.boundary.kind!=='waiting')throw new Error();
  const closing=game.runtime.close(ref);
  await assert.rejects(game.runtime.advance({game:ref,boundaryId:update.boundary.boundaryId,input:action(update.boundary.pending,{kind:'bid',value:3})}),error=>GameErrorSchema.parse(error).code==='game_busy');
  release();
  await assert.rejects(closing,error=>GameErrorSchema.parse(error).code==='execution_uncertain');
  await assert.rejects(game.runtime.inspect(ref),error=>GameErrorSchema.parse(error).code==='game_faulted');
  await game.runtime.close(ref);
  await game.runtime.close(ref);
  await assert.rejects(game.runtime.inspect(ref),error=>GameErrorSchema.parse(error).code==='game_not_found');
});
