import { z } from 'zod';
import type { GameContext, GameModule, GameSchemas, GameQueryCapability, InteractionOptions, RequestArgs, QueryCall, DescribeCall, GameRef, BoundaryId, FiniteTableGuard, LoadedGame, CompiledGame } from '@bear-forge/contracts';
const schema={left:{request:z.strictObject({left:z.number()}),input:z.number(),context:z.strictObject({left:z.number()}),description:z.never()},right:{request:z.strictObject({right:z.string()}),input:z.string(),context:z.strictObject({right:z.string()}),description:z.strictObject({limit:z.number()})}};
type T={[K in keyof typeof schema]:{[F in keyof (typeof schema)[K]]:z.infer<(typeof schema)[K][F]>}};
type G={setup:null;state:null;interactions:T;event:null;result:null};
type Q={left:{input:number;output:number};right:{input:string;output:string}};
export const checkPairs=(ctx:GameContext<G>,queries:GameQueryCapability<Q>,options:InteractionOptions<T>,game:GameRef,boundaryId:BoundaryId,key:'left'|'right',args:RequestArgs<T>,query:QueryCall<Q>,describe:DescribeCall<T>)=>{
  const n:Promise<number>=ctx.input.request('left',{left:1});
  const s:Promise<string>=ctx.input.request('right',{right:'yes'});
  const both:Promise<number|string>=ctx.input.request(...args);
  const q:Promise<number|string>=queries.query(query);
  options.describe(describe);
  // @ts-expect-error explicit old widened key cannot stand in for a tuple
  ctx.input.request<'left'|'right'>('left',{right:'wrong'});
  // @ts-expect-error even the widest accepted tuple union preserves pairs
  ctx.input.request<RequestArgs<T>>('left',{right:'wrong'});
  // @ts-expect-error independent union key and payload are not correlated
  ctx.input.request(key,{right:'wrong'});
  // @ts-expect-error wrong literal pairing
  ctx.input.request('left',{right:'wrong'});
  // @ts-expect-error widening the whole query union does not add cross products
  queries.query<QueryCall<Q>>({target:{kind:'instance',game},name:'left',args:'wrong'});
  // @ts-expect-error independent key remains unsafe
  queries.query({target:{kind:'instance',game},name:key,args:'wrong'});
  // @ts-expect-error widening the descriptor union does not add cross products
  options.describe<DescribeCall<T>>({target:{kind:'instance',game},boundaryId,type:'left',context:{right:'wrong'}});
  options.validate({target:{kind:'instance',game},boundaryId,reply:{type:'left',data:1}});
  // @ts-expect-error validation also uses paired replies
  options.validate({target:{kind:'instance',game},boundaryId,reply:{type:'left',data:'wrong'}});
  return {n,s,both,q};
};

export const empty:GameModule<{setup:null;state:null;event:null;result:null;interactions:{}},{}>={schemas:{...domain,interactions:{}},initialize:()=>null,run:async()=>null,inputs:{},queries:{}};
