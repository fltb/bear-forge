import type { Execution, ExecutionCapture, Program, RandomCapability, StateTransitionCapability, GameRuntime, EvaluationCapability } from '@bear-forge/contracts';
import type { Setup, Frame, Input, Pending, AuditEvent, Result, DouDizhuRuntime } from '../../games/doudizhu/src/index.ts';
// A runtime does not need observation, reward, facts or persistence to satisfy the contract.
export const runtimeConsumer = (runtime:GameRuntime<Setup,Input,Pending,AuditEvent,Result>):DouDizhuRuntime=>runtime;
export const coreConsumer = (core:Execution<Setup,Frame,Input,Frame>)=>core;
export const captureConsumer = (capture:ExecutionCapture<Setup,Frame,Input,Frame>)=>capture;
// Pure search can use numbers, with no actor, opponent, Session, or GameRef.
export const numericTransition:StateTransitionCapability<number,number,number> = {
  inspect:async state=>state,
  transition:async ({state,input})=>({state:state+input,view:state+input}),
};
export const vectorEvaluation:EvaluationCapability<number,{},number[]> = {evaluate:async ({source})=>[source,-source]};
export const checks = (runtime:DouDizhuRuntime,core:Execution<Setup,Frame,Input,Frame>)=>{
  // @ts-expect-error reward is not a compulsory runtime capability
  runtime.reward;
  // @ts-expect-error snapshots are a separate capability
  core.save;
  // @ts-expect-error concrete game inputs cannot be replaced by arbitrary payload envelopes
  const input:Input={schema:'anything',value:{}};
  return input;
};
export const plainProgram:Program<number,number,number,number,{}> = async (input,context)=>input+await context.request(input);
export const randomProgram:Program<number,number,number,number,RandomCapability> = async (input,context)=>input+await context.integer({minInclusive:0,maxExclusive:2});
